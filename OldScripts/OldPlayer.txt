func spawn_mob():
	if mob_count >= MAX_MOBS:
		return
		
	var new_mob = preload("res://scenes/enemys/weasel/weasel_enemy.tscn").instantiate()
	%PathFollow2D.progress_ratio = randf()
	new_mob.global_position = %PathFollow2D.global_position

	mob_count += 1
